cmake_minimum_required(VERSION 3.22)
project(SuperTux3DS LANGUAGES C CXX)
# MSYS make cannot read native GCC depfiles containing an unescaped C: drive.
# Let CMake scan includes using its own platform-aware dependency scanner.
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "MSYS")
  set(CMAKE_C_DEPENDS_USE_COMPILER FALSE)
  set(CMAKE_CXX_DEPENDS_USE_COMPILER FALSE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fdata-sections -ffunction-sections -fno-strict-aliasing")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fdata-sections -ffunction-sections -fno-strict-aliasing")
get_filename_component(ST_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VENDOR "${ST_ROOT}/3ds/vendor")

set(SDL_SYSTEM_ICONV OFF CACHE BOOL "" FORCE)
set(HAVE_ICONV 0 CACHE INTERNAL "Use SDL builtin iconv on newlib" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/SDL" sdl)
add_library(SDL2 ALIAS SDL2-static)

set(SDL2IMAGE_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_DEPS_SHARED OFF CACHE BOOL "" FORCE)
foreach(format AVIF JXL TIF WEBP)
  set(SDL2IMAGE_${format} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${VENDOR}/SDL_image" sdl-image)
set(SDL2MIXER_INSTALL OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_DEPS_SHARED OFF CACHE BOOL "" FORCE)
set(SDL2MIXER_VORBIS TREMOR CACHE STRING "" FORCE)
set(SDL2MIXER_VORBIS_TREMOR_SHARED OFF CACHE BOOL "" FORCE)
set(tremor_LIBRARY "${DEVKITPRO}/portlibs/3ds/lib/libvorbisidec.a" CACHE FILEPATH "" FORCE)
set(tremor_INCLUDE_PATH "${DEVKITPRO}/portlibs/3ds/include" CACHE PATH "" FORCE)
set(tremor_LINK_LIBRARIES "${DEVKITPRO}/portlibs/3ds/lib/libogg.a" CACHE STRING "" FORCE)
foreach(codec CMD FLAC GME MOD MP3 MIDI OPUS WAVPACK)
  set(SDL2MIXER_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${VENDOR}/SDL_mixer" sdl-mixer)

set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/freetype" freetype)
add_library(Freetype ALIAS freetype)

add_library(ZLIB STATIC IMPORTED GLOBAL)
set_target_properties(ZLIB PROPERTIES IMPORTED_LOCATION "${DEVKITPRO}/portlibs/3ds/lib/libz.a"
  INTERFACE_INCLUDE_DIRECTORIES "${DEVKITPRO}/portlibs/3ds/include")
add_library(PNG STATIC IMPORTED GLOBAL)
set_target_properties(PNG PROPERTIES IMPORTED_LOCATION "${DEVKITPRO}/portlibs/3ds/lib/libpng.a"
  INTERFACE_INCLUDE_DIRECTORIES "${DEVKITPRO}/portlibs/3ds/include" INTERFACE_LINK_LIBRARIES ZLIB)
# Build the included, devkitPro-patched PhysicsFS source. No local binary-only
# 3ds/deps directory is needed when someone clones the port.
set(PHYSFS_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(PHYSFS_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(PHYSFS_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(PHYSFS_BUILD_DOCS OFF CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/physfs" physfs)
add_library(PhysFS ALIAS physfs-static)

set(FMT_INSTALL OFF CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${VENDOR}/fmt" fmt)
add_library(glm INTERFACE)
target_include_directories(glm INTERFACE "${VENDOR}/glm")
target_compile_definitions(glm INTERFACE GLM_ENABLE_EXPERIMENTAL)

set(DISABLE_DYNAMIC ON CACHE BOOL "" FORCE)
set(SQ_DISABLE_INTERPRETER ON CACHE BOOL "" FORCE)
set(SQ_DISABLE_INSTALLER ON CACHE BOOL "" FORCE)
set(SQ_ROOT "${ST_ROOT}/external/simplesquirrel/libs/squirrel")
file(GLOB SQ_SOURCES "${SQ_ROOT}/squirrel/*.cpp" "${SQ_ROOT}/sqstdlib/*.cpp")
add_library(squirrel_static STATIC ${SQ_SOURCES})
target_include_directories(squirrel_static PUBLIC "${SQ_ROOT}/include")
file(GLOB SSQ_SOURCES "${ST_ROOT}/external/simplesquirrel/source/*.cpp")
add_library(simplesquirrel_static STATIC ${SSQ_SOURCES})
target_include_directories(simplesquirrel_static PUBLIC "${ST_ROOT}/external/simplesquirrel/include")
target_link_libraries(simplesquirrel_static PUBLIC squirrel_static)
add_subdirectory("${ST_ROOT}/external/sexp-cpp" sexp)
add_subdirectory("${ST_ROOT}/external/tinygettext" tinygettext)
add_subdirectory("${ST_ROOT}/external/SDL_ttf" sdl-ttf)
add_subdirectory("${ST_ROOT}/external/SDL_SavePNG" savepng)
add_subdirectory("${ST_ROOT}/external/partio_zip" partio)
add_subdirectory("${ST_ROOT}/external/findlocale" findlocale)
add_subdirectory("${ST_ROOT}/external/obstack" obstack)

set(SUPERTUX_PACKAGE_VERSION "0.7.0-3ds-experimental")
set(SUPERTUX_PACKAGE_VERSION_TAG "0.7.0")
set(LOGO_FILE "images/engine/menu/logo.png")
set(BUILD_DATA_DIR "sdmc:/3ds/supertux070/data")
set(BUILD_CONFIG_DATA_DIR "sdmc:/3ds/supertux070/data")
set(INSTALL_SUBDIR_SHARE "data")
set(INSTALL_SUBDIR_BIN "")
set(HIDE_NONMOBILE_OPTIONS ON)
configure_file("${ST_ROOT}/config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/config.h")
configure_file("${ST_ROOT}/version.h.in" "${CMAKE_CURRENT_BINARY_DIR}/version.h")

file(GLOB ENGINE_SOURCES "${ST_ROOT}/src/*/*.cpp" "${ST_ROOT}/src/supertux/menu/*.cpp"
  "${ST_ROOT}/src/video/sdl/*.cpp" "${ST_ROOT}/src/video/null/*.cpp")
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/audio/")
list(FILTER ENGINE_SOURCES EXCLUDE REGEX "/sdl_painter.cpp$")
add_library(supertux_core OBJECT ${ENGINE_SOURCES} "${ST_ROOT}/src/audio/dummy_sound_source.cpp"
  "${ST_ROOT}/3ds/source/platform.cpp" mixer_audio.cpp lazy_texture.cpp gpu_renderer.cpp gpu_painter.cpp
  performance.cpp packed_assets.cpp texture_prefetch.cpp native_hud.cpp debug_unlock.cpp loading.cpp native_controls.cpp)
target_include_directories(supertux_core BEFORE PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
  "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}" "${ST_ROOT}/src" "${ST_ROOT}/3ds/source")
target_compile_definitions(supertux_core PUBLIC SDL_MAIN_HANDLED GLM_ENABLE_EXPERIMENTAL)
target_compile_options(supertux_core PUBLIC -fexceptions -frtti)
target_link_libraries(supertux_core PUBLIC SDL2_image SDL2_ttf SDL2_mixer SDL2 simplesquirrel_static
  sexp tinygettext SDL_SavePNG PartioZip FindLocale obstack PhysFS fmt glm PNG ZLIB citro2d citro3d)
add_executable(supertux070 main.cpp native_heap.c)
foreach(target supertux070)
  target_link_libraries(${target} PRIVATE supertux_core SDL_SavePNG PartioZip FindLocale obstack)
  target_link_options(${target} PRIVATE -Wl,--gc-sections "-Wl,-Map,${target}.map")
  set_target_properties(${target} PROPERTIES SUFFIX ".elf")
endforeach()

# Standalone VM regression probe, excluded from ordinary builds/packages.
add_executable(supertux-vm-probe EXCLUDE_FROM_ALL vm_probe.cpp)
target_link_libraries(supertux-vm-probe PRIVATE supertux_core SDL_SavePNG PartioZip FindLocale obstack)
target_link_options(supertux-vm-probe PRIVATE -Wl,--gc-sections)
set_target_properties(supertux-vm-probe PROPERTIES SUFFIX ".elf")
